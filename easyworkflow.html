Drop-In REST API Integration

Trigger envelope creation, assign signers, and receive signed document webhooks in under 10 lines of code.

POST /v1/envelopes/send // node.js 
// 1. Initialize EasyWorkFlow Client
import { EasyWorkFlow } from '@easylabs/easyworkflow';
const easyworkflow = new EasyWorkFlow({ apiKey: process.env.EASYWORKFLOW_API_KEY });

// 2. Dispatch Multi-Party Contract Envelope
const envelope = await easyworkflow.envelopes.createAndSend({
  templateId: 'tmpl_enterprise_msa_2026',
  title: 'Enterprise Master Services Agreement — Acme Corp',
  signers: [
    { email: 'legal@acme.com', role: 'Signer', order: 1 },
    { email: 'cfo@acme.com', role: 'Approver', order: 2 }
  ],
  webhookUrl: 'https://api.yourdomain.com/webhooks/easyworkflow'
});

console.log(`Envelope Dispatched! SHA-256 Tracking: ${envelope.trackingId}`);
